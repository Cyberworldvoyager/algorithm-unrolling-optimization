\begin{abstract}
This project explores \textbf{Algorithm Unrolling} techniques for the LASSO sparse coding problem. We transform the classical ISTA algorithm into a learnable neural network.

\textbf{Core contributions}:
\begin{enumerate}
    \item Propose \textbf{LISTA-Momentum} with learnable momentum to solve generalization
    \item Design \textbf{dimension-agnostic LISTA} that handles arbitrary input dimensions
    \item Provide \textbf{spectral radius stability verification} and \textbf{parameter matrix analysis}
\end{enumerate}

\textbf{Results}: LISTA-Momentum achieves only 1.06× generalization degradation, outperforming ISTA on both ID and OOD. Dimension-agnostic LISTA achieves 3.77-8.72× performance improvement within training dimension range.

\bigskip
\noindent\textbf{Keywords:} Algorithm Unrolling, LISTA, LASSO, Dimension-Agnostic, Generalization
\end{abstract}

\section{Introduction}
\label{sec:intro}

\subsection{Background}

LASSO (Least Absolute Shrinkage and Selection Operator) is a core optimization problem in sparse signal recovery:
\begin{equation}
    \min_x \frac{1}{2} \|Ax - b\|^2 + \lambda \|x\|_1
\end{equation}

The classical ISTA (Iterative Shrinkage-Thresholding Algorithm) solves this iteratively:
\begin{equation}
    x_{k+1} = \text{SoftThreshold}(x_k - \eta A^T(Ax_k - b), \eta\lambda)
\end{equation}

\textbf{Problem}: ISTA requires many iterations to converge, leading to high computational costs.

\textbf{Solution}: Unroll ISTA into a neural network and learn better parameters.

\subsection{Main Contributions}

\begin{enumerate}
    \item \textbf{LISTA-Momentum}: Propose LISTA with learnable momentum to solve generalization
    \item \textbf{Dimension-agnostic LISTA}: Design dimension-agnostic architecture for arbitrary input dimensions
    \item \textbf{Theoretical analysis}: Provide spectral radius stability verification and parameter matrix analysis
\end{enumerate}

% ==================== 预备知识 ====================
\section{Preliminaries}
\label{sec:prelim}

\subsection{Proximal Operator}

The proximal operator is a key tool for solving non-smooth optimization problems:
\begin{equation}
    \text{prox}_{\lambda f}(v) = \arg\min_x \left( f(x) + \frac{1}{2\lambda} \|x - v\|^2 \right)
\end{equation}

For L1 regularization, the proximal operator is the soft thresholding function:
\begin{equation}
    \text{prox}_{\lambda|\cdot|_1}(v)_i = \text{sign}(v_i) \max(|v_i| - \lambda, 0)
\end{equation}

\subsection{ISTA Algorithm}

ISTA solves the LASSO problem:
\begin{equation}
    x_{k+1} = \text{SoftThreshold}(x_k - \eta A^T(Ax_k - b), \eta\lambda)
\end{equation}

\textbf{Convergence rate}: $\|x_k - x^*\| = O(1/k)$

\subsection{FISTA Algorithm}

FISTA accelerates ISTA with Nesterov momentum:
\begin{align}
    y_k &= x_k + \frac{k-1}{k+2}(x_k - x_{k-1}) \\
    x_{k+1} &= \text{SoftThreshold}(y_k - \eta A^T(Ay_k - b), \eta\lambda)
\end{align}

\textbf{Convergence rate}: $\|x_k - x^*\| = O(1/k^2)$

\section{Unrolling Methods}
\label{sec:methods}

\subsection{LISTA (Gregor \& LeCun, 2010)}

\textbf{Core idea}: Unroll $T$ ISTA iterations into $T$ layers with learnable parameters $W_1, W_2, \theta$:
\begin{equation}
    x_{k+1} = \sigma(W_1 b + W_2 x_k; \theta_k)
\end{equation}

\textbf{Problem}: $W_1, W_2$ are fixed, cannot generalize to new $A$.

\subsection{LISTA-CP (Chen et al., 2018)}

\textbf{Core improvement}: Couple weights using $B$:
\begin{equation}
    W_1 = \eta B, \quad W_2 = I - \eta BA
\end{equation}

\textbf{Advantage}: Parameters reduced from $O(nm + n^2)$ to $O(nm)$

\textbf{Problem}: $B$ is tied to training matrix $A_{train}$, cannot generalize to new $A$.

\subsection{LISTA-Momentum (Ours)}

\textbf{Core innovation}: Introduce learnable momentum to solve generalization.

\subsubsection{Mathematical Formulation}

\begin{equation}
\boxed{
\begin{aligned}
y_k &= x_k + \sigma(\beta) \cdot (x_k - x_{k-1}) & \text{(momentum extrapolation)} \\
g_k &= A^T(A y_k - b) & \text{(gradient with current A)} \\
x_{k+1} &= \text{SoftThreshold}(y_k - \eta \cdot W g_k; \theta) & \text{(learnable transformation)}
\end{aligned}
}
\end{equation}

\subsubsection{Parameter Description}

\centering
\begin{tabular}{llll}
\toprule
\textbf{Parameter} & \textbf{Dimension} & \textbf{Role} & \textbf{Initialization} \\
\midrule
$W$ & $n \times n$ & Gradient transformation & Identity matrix $I$ \\
$\eta$ & scalar & Step size & 0.1 \\
$\beta$ & scalar & Momentum & 0.0 (sigmoid → 0.5) \\
$\theta$ & scalar & Threshold & 0.1 \\
\bottomrule
\end{tabular}

\label{tab:params}
\end{table}

\subsubsection{Key Design}

\begin{enumerate}
    \item \textbf{Don't embed A in weights}: Compute gradient with current A at each layer
    \item \textbf{Learn general gradient transformation W}: Independent of A
    \item \textbf{Learnable momentum $\beta$}: Adaptive acceleration
\end{enumerate}

\subsubsection{PyTorch Implementation}

\begin{lstlisting}[language=Python]
class LISTAMomentumLayer(nn.Module):
    def __init__(self, m, n):
        super().__init__()
        # Learnable gradient transformation W
        self.W = nn.Linear(n, n, bias=False)
        nn.init.eye_(self.W.weight)  # Initialize as identity

        # Learnable step size and momentum
        self.eta = nn.Parameter(torch.tensor(0.1))
        self.beta = nn.Parameter(torch.tensor(0.0))

        # Learnable threshold
        self.threshold = nn.Parameter(torch.tensor(0.1))

    def forward(self, b, x, x_prev, A):
        # Momentum extrapolation
        beta = torch.sigmoid(self.beta)
        y = x + beta * (x - x_prev)

        # Gradient with current A
        Ay = torch.bmm(A, y.unsqueeze(-1)).squeeze(-1)
        residual = Ay - b
        grad = torch.bmm(A.transpose(1,2), residual.unsqueeze(-1)).squeeze(-1)

        # Learnable transformation
        transformed_grad = self.W(grad)
        z = y - self.eta * transformed_grad

        # Soft thresholding
        return torch.sign(z) * torch.maximum(
            torch.abs(z) - self.threshold, torch.zeros_like(z))
\end{lstlisting}

\subsection{Dimension-Agnostic LISTA (Ours)}

\textbf{Core problem}: LISTA-Momentum uses \texttt{nn.Linear(n, n)} for gradient transformation $W$, which is fixed to dimension $n$. When problem dimension changes, retraining is needed.

\textbf{Core innovation}: Design dimension-agnostic architecture that handles arbitrary input dimensions.

\subsubsection{Design Philosophy}

\textbf{Key insight}: The gradient transformation $W$ should be independent of dimension $n$.

\textbf{Solution}:
\begin{enumerate}
    \item \textbf{LayerNorm normalization}: Eliminate dimension effects
    \item \textbf{Shared MLP transformation}: Same transform for each element
    \item \textbf{Scale factor}: Restore pre-normalization scale
\end{enumerate}

\subsubsection{Mathematical Formulation}

\begin{equation}
\boxed{
\begin{aligned}
y_k &= x_k + \sigma(\beta) \cdot (x_k - x_{k-1}) & \text{(momentum extrapolation)} \\
g_k &= A^T(A y_k - b) & \text{(gradient with current A)} \\
\bar{g}_k &= \text{LayerNorm}(g_k) & \text{(normalization)} \\
\hat{g}_k &= \text{MLP}(\bar{g}_k) & \text{(shared transformation)} \\
x_{k+1} &= \text{SoftThreshold}(y_k - \eta \cdot \hat{g}_k \cdot \text{std}(g_k); \theta) & \text{(restore scale)}
\end{aligned}
}
\end{equation}

\subsubsection{Key Components}

\textbf{1. LayerNorm Normalization}:
\begin{equation}
    \bar{g} = \frac{g - \mu(g)}{\sigma(g) + \epsilon}
\end{equation}
Eliminates the effect of dimension $n$, making transformation dimension-independent.

\textbf{2. Shared MLP Transformation}:
\begin{lstlisting}[language=Python]
self.transform = nn.Sequential(
    nn.Linear(1, hidden_dim),
    nn.GELU(),
    nn.Linear(hidden_dim, 1),
)
\end{lstlisting}
Applies the same transformation to each element, parameters independent of dimension $n$.

\textbf{3. Scale Factor}:
\begin{equation}
    \hat{g} = \text{MLP}(\bar{g}) \cdot \sigma(g)
\end{equation}
Restores the pre-normalization scale.

\subsubsection{Complete PyTorch Implementation}

\begin{lstlisting}[language=Python]
class DimensionAgnosticLayer(nn.Module):
    """Dimension-agnostic LISTA-Momentum layer.

    Key design:
    1. LayerNorm to normalize gradient (eliminate dimension effects)
    2. Shared MLP transformation (dimension-independent)
    3. Learnable scale factor (restore pre-normalization scale)
    """

    def __init__(self, hidden_dim=32):
        super().__init__()

        # Gradient transformation network (dimension-agnostic)
        self.transform = nn.Sequential(
            nn.Linear(1, hidden_dim),
            nn.GELU(),
            nn.Linear(hidden_dim, hidden_dim),
            nn.GELU(),
            nn.Linear(hidden_dim, 1),
        )

        # Learnable step size and momentum
        self.eta = nn.Parameter(torch.tensor(0.1))
        self.beta = nn.Parameter(torch.tensor(0.0))

        # Learnable threshold
        self.threshold = nn.Parameter(torch.tensor(0.1))

        # Learnable scale factor
        self.scale = nn.Parameter(torch.tensor(1.0))

    def forward(self, b, x, x_prev, A):
        # Momentum extrapolation
        beta = torch.sigmoid(self.beta)
        y = x + beta * (x - x_prev)

        # Gradient with current A: g = A^T (A y - b)
        Ay = torch.bmm(A, y.unsqueeze(-1)).squeeze(-1)
        residual = Ay - b
        grad = torch.bmm(A.transpose(1, 2), residual.unsqueeze(-1)).squeeze(-1)

        # Dimension-agnostic transformation
        batch_size, n = grad.shape

        # 1. Normalization (eliminate dimension effects)
        grad_mean = grad.mean(dim=-1, keepdim=True)
        grad_std = grad.std(dim=-1, keepdim=True) + 1e-6
        grad_normalized = (grad - grad_mean) / grad_std

        # 2. Element-wise transformation (shared weights)
        grad_flat = grad_normalized.reshape(-1, 1)  # (batch*n, 1)
        transformed_flat = self.transform(grad_flat)  # (batch*n, 1)
        transformed_normalized = transformed_flat.reshape(batch_size, n)  # (batch, n)

        # 3. Restore scale + residual connection
        transformed_grad = (transformed_normalized + grad_normalized) * grad_std * self.scale

        # Update
        z = y - self.eta * transformed_grad

        # Soft thresholding
        return torch.sign(z) * torch.maximum(
            torch.abs(z) - self.threshold, torch.zeros_like(z))


class DimensionAgnosticLISTA(nn.Module):
    """Dimension-agnostic LISTA-Momentum.

    Can handle different A matrix dimensions without retraining.
    """

    def __init__(self, T=10, hidden_dim=32):
        super().__init__()
        self.T = T
        self.layers = nn.ModuleList([
            DimensionAgnosticLayer(hidden_dim) for _ in range(T)
        ])

    def forward(self, b, A, x0=None):
        batch_size = b.shape[0]
        if A.dim() == 2:
            A = A.unsqueeze(0).expand(batch_size, -1, -1)

        n = A.shape[2]
        x = x0 if x0 is not None else torch.zeros(batch_size, n, device=b.device)
        x_prev = x.clone()

        for layer in self.layers:
            x_new = layer(b, x, x_prev, A)
            x_prev = x
            x = x_new

        return x
\end{lstlisting}
